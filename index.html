function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); var result = []; // 第 1 列為標頭 (ID, 姓名, 單位, 日期, 時間, 狀態) for (var i = 1; i < data.length; i++) { result.push({ id: String(data[i][0] || '').replace("'", ""), name: String(data[i][1] || ''), unit: String(data[i][2] || ''), date: formatDateString(data[i][3]), // 確保讀取時也是正確的日期字串 time: String(data[i][4] || ''), status: String(data[i][5] || '') }); } var callback = e.parameter.callback; var jsonOutput = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + '(' + jsonOutput + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonOutput) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); // 避免多筆同時寫入造成的編號衝突 try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var param = JSON.parse(e.postData.contents); var action = param.action; if (action === "book") { var bookingDate = String(param.date || '').trim(); // 傳入的日期 (如: 2026-09-04) var bookingName = String(param.name || '').trim(); var bookingUnit = String(param.unit || '').trim(); var bookingTime = String(param.time || '').trim(); if (!bookingDate) { return ContentService.createTextOutput(JSON.stringify({ result: "error", message: "未提供有效的日期" })).setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); var targetId = ""; // 1. 檢查試算表中是否已經有「相同日期」的預約紀錄，如果有就使用同一個編號 for (var i = 1; i < data.length; i++) { var rowDate = formatDateString(data[i][3]); var rowId = String(data[i][0] || '').trim(); if (rowDate === bookingDate && rowId !== "") { targetId = rowId; // 找到了相同日期的既有編號 break; } } // 2. 若該日期尚未有任何紀錄，則根據目前時間建立一組新的編號 (格式: 'YYMMDDHHmm) if (!targetId) { var now = new Date(); var timeZone = Session.getScriptTimeZone(); targetId = "'" + Utilities.formatDate(now, timeZone, "yyMMddHHmm"); } else { // 確保帶有單引號單頭 if (!targetId.startsWith("'")) { targetId = "'" + targetId; } } // 3. 確保日期欄位加上單引號轉為純字串格式，防止試算表自動清空或格式跑掉 var formattedBookingDate = "'" + bookingDate; // 4. 寫入 Google 試算表：[ID, 姓名, 單位, 日期, 時間, 狀態] sheet.appendRow([targetId, bookingName, bookingUnit, formattedBookingDate, bookingTime, "已預約"]); return ContentService.createTextOutput(JSON.stringify({ result: "success", id: targetId.replace("'", "") })).setMimeType(ContentService.MimeType.JSON); } else if (action === "cancel") { var targetId = String(param.id || '').trim(); var targetName = String(param.name || '').trim(); var data = sheet.getDataRange().getValues(); var found = false; for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0]).replace("'", "").trim(); var rowName = String(data[i][1]).trim(); var rowStatus = String(data[i][5]).trim(); if (rowId === targetId && rowName === targetName && rowStatus === "已預約") { // 將狀態改為已取消 sheet.getRange(i + 1, 6).setValue("已取消"); found = true; break; } } if (found) { return ContentService.createTextOutput(JSON.stringify({ result: "success" })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ result: "error", message: "找不到該預約編號或預約姓名不符" })).setMimeType(ContentService.MimeType.JSON); } } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ result: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // 輔助函式：確保讀取與比對日期的格式為 YYYY-MM-DD function formatDateString(val) { if (!val) return ''; if (val instanceof Date) { var yyyy = val.getFullYear(); var mm = String(val.getMonth() + 1).padStart(2, '0'); var dd = String(val.getDate()).padStart(2, '0'); return yyyy + '-' + mm + '-' + dd; } return String(val).replace("'", "").trim(); }
